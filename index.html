function doGet(e) { var lat = parseFloat(e.parameter.lat); var lng = parseFloat(e.parameter.lng); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Sheet1"); // เปลี่ยนชื่อชีตตามจริงของคุณหมอถ้าไม่ใช่ Sheet1 var data = sheet.getDataRange().getValues(); var minDistance = Infinity; var closestLocation = "ไม่พบสถานที่ใกล้เคียง"; // วนลูปหาพิกัดที่ใกล้ที่สุดจากตาราง (สมมติว่า Column B = ชื่อสถานที่, Column D = พิกัด เช่น "13.7320317,100.588621") for (var i = 1; i < data.length; i++) { var placeName = data[i][1]; // Column B (Index 1) var coordsStr = data[i][3]; // Column D (Index 3) if (coordsStr && coordsStr.includes(",")) { var parts = coordsStr.split(","); var targetLat = parseFloat(parts[0].trim()); var targetLng = parseFloat(parts[1].trim()); // คำนวณระยะทางแบบ Haversine (หน่วยเป็นเมตร) var distance = getDistanceFromLatLonInMeters(lat, lng, targetLat, targetLng); if (distance < minDistance) { minDistance = distance; closestLocation = placeName; } } } // ถ้าระยะห่างเกิน 500 เมตร อาจจะแสดงเตือนหรือแสดงชื่อที่ใกล้ที่สุด return ContentService.createTextOutput(JSON.stringify({ location: closestLocation, distanceMeters: Math.round(minDistance) })).setMimeType(ContentService.MimeType.JSON); } // ฟังก์ชันคำนวณระยะทาง Haversine Formula function getDistanceFromLatLonInMeters(lat1, lon1, lat2, lon2) { var R = 6371000; // Radius of the earth in meters var dLat = deg2rad(lat2 - lat1); var dLon = deg2rad(lon2 - lon1); var a = Math.sin(dLat/2) * Math.sin(dLat/2) + Math.cos(deg2rad(lat1)) * Math.cos(deg2rad(lat2)) * Math.sin(dLon/2) * Math.sin(dLon/2) ; var c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); var d = R * c; // Distance in meters return d; } function deg2rad(deg) { return deg * (Math.PI/180); }
